function renderDashboard(rows) { const container = document.getElementById("dailyDataContainer"); if (!container) return; container.innerHTML = ""; let totalCash = 0; let totalCard = 0; rows.forEach(row => { const dayName = row[0] ? row[0].toString().trim() : ""; let rawDate = row[1] ? row[1].toString().trim() : ""; const cashVal = row[2]; const cardVal = row[3]; // 1. Skip rows without valid date strings, headers, or explicit summary rows if (!rawDate || dayName.toLowerCase().includes("month total") || rawDate.toLowerCase().includes("date")) { return; } // 2. Clean numeric values const cleanNum = (val) => parseFloat(String(val || 0).replace(/[^0-9.-]+/g, "")) || 0; const cash = cleanNum(cashVal); const card = cleanNum(cardVal); // 3. Skip weekly subtotal rows (lack day name and don't contain standard ISO/Date indicators) const isStandardDate = rawDate.includes("T") || rawDate.includes("/") || rawDate.includes("-"); if (!dayName && (cash > 0 || card > 0) && !isStandardDate) { return; } // 4. Format ISO timestamps into clean DD/MM/YYYY dates let displayDate = rawDate; if (rawDate.includes("T")) { const parsedDate = new Date(rawDate); if (!isNaN(parsedDate.getTime())) { displayDate = parsedDate.toLocaleDateString("en-GB"); } } const dailyTotal = cash + card; // Skip empty non-trading days if (dailyTotal === 0 && !dayName) { return; } totalCash += cash; totalCard += card; // 5. Safely read category columns (Columns E, F, G) const drink = cleanNum(row[4]); const food = cleanNum(row[5]); const deli = cleanNum(row[6]); // 6. Build card element with full HTML structure const cardElement = document.createElement("div"); cardElement.className = "daily-card"; cardElement.innerHTML = `
${dayName} ${displayDate} £${dailyTotal.toFixed(2)}

Cash: £${cash.toFixed(2)} | Card: £${card.toFixed(2)}

DRINK: £${drink.toFixed(2)}
FOOD: £${food.toFixed(2)}
DELI: £${deli.toFixed(2)}
`; container.appendChild(cardElement); }); updateTotals(totalCash, totalCard); } function updateTotals(cash, card) { if (document.getElementById("totalRev")) document.getElementById("totalRev").innerText = `£${(cash + card).toFixed(2)}`; if (document.getElementById("totalCash")) document.getElementById("totalCash").innerText = `£${cash.toFixed(2)}`; if (document.getElementById("totalCard")) document.getElementById("totalCard").innerText = `£${card.toFixed(2)}`; }
