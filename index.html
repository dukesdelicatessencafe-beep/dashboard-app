function renderDashboard(rows) { const container = document.getElementById("dailyDataContainer"); if (!container) return; container.innerHTML = ""; let totalCash = 0; let totalCard = 0; rows.forEach(row => { // Col A = Day Name (row[0]), Col B = Date (row[1]), Col C = Cash (row[2]), Col D = Card (row[3]) const dayName = row[0] ? row[0].toString().trim() : ""; const dateStr = row[1] ? row[1].toString().trim() : ""; // Skip empty rows, header rows, or summary/monthly total rows if (!dateStr || dateStr.toLowerCase().includes("month total") || dateStr.toLowerCase().includes("date") || dateStr.toLowerCase().includes("catorgory")) { return; } // Strips out £ symbols, spaces, and commas const cleanNum = (val) => parseFloat(String(val || 0).replace(/[^0-9.-]+/g, "")) || 0; const cash = cleanNum(row[2]); const card = cleanNum(row[3]); const dailyTotal = cash + card; totalCash += cash; totalCard += card; // Safely reads categories (DRINK, FOOD, DELI, etc.) const drink = cleanNum(row[4]); const food = cleanNum(row[5]); const deli = cleanNum(row[6]); const cardElement = document.createElement("div"); cardElement.className = "daily-card"; cardElement.innerHTML = `
${dayName} ${dateStr} £${dailyTotal.toFixed(2)}

Cash: £${cash.toFixed(2)} | Card: £${card.toFixed(2)}

DRINK: £${drink.toFixed(2)}
FOOD: £${food.toFixed(2)}
DELI: £${deli.toFixed(2)}
`; container.appendChild(cardElement); }); updateTotals(totalCash, totalCard); } function updateTotals(cash, card) { if (document.getElementById("totalRev")) document.getElementById("totalRev").innerText = `£${(cash + card).toFixed(2)}`; if (document.getElementById("totalCash")) document.getElementById("totalCash").innerText = `£${cash.toFixed(2)}`; if (document.getElementById("totalCard")) document.getElementById("totalCard").innerText = `£${card.toFixed(2)}`; }
