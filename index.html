function renderDashboard(rows) { const container = document.getElementById("dailyDataContainer"); if (!container) return; container.innerHTML = ""; let totalCash = 0; let totalCard = 0; rows.forEach(row => { const dayName = row[0] ? row[0].toString().trim() : ""; let rawDate = row[1] ? row[1].toString().trim() : ""; const cashVal = row[2]; const cardVal = row[3]; // 1. Skip rows without valid sales numbers or rows explicitly marked as totals if ((cashVal === "" && cardVal === "") || !rawDate || dayName.toLowerCase().includes("month total")) { return; } // 2. Format ISO timestamps (e.g., 2023-04-11T07:00:00.000Z) into readable dates (11/04/2023) let displayDate = rawDate; if (rawDate.includes("T")) { const parsedDate = new Date(rawDate); if (!isNaN(parsedDate.getTime())) { displayDate = parsedDate.toLocaleDateString("en-GB"); // DD/MM/YYYY } } // 3. Clean numbers and calculate daily total const cleanNum = (val) => parseFloat(String(val || 0).replace(/[^0-9.-]+/g, "")) || 0; const cash = cleanNum(cashVal); const card = cleanNum(cardVal); const dailyTotal = cash + card; totalCash += cash; totalCard += card; // 4. Extract Category values (Columns E, F, G) const drink = cleanNum(row[4]); const food = cleanNum(row[5]); const deli = cleanNum(row[6]); // 5. Build and render the card UI const cardElement = document.createElement("div"); cardElement.className = "daily-card"; cardElement.innerHTML = `
${dayName} ${displayDate} £${dailyTotal.toFixed(2)}

Cash: £${cash.toFixed(2)} | Card: £${card.toFixed(2)}

DRINK: £${drink.toFixed(2)}
FOOD: £${food.toFixed(2)}
DELI: £${deli.toFixed(2)}
`; container.appendChild(cardElement); }); updateTotals(totalCash, totalCard); } function updateTotals(cash, card) { if (document.getElementById("totalRev")) document.getElementById("totalRev").innerText = `£${(cash + card).toFixed(2)}`; if (document.getElementById("totalCash")) document.getElementById("totalCash").innerText = `£${cash.toFixed(2)}`; if (document.getElementById("totalCard")) document.getElementById("totalCard").innerText = `£${card.toFixed(2)}`; }
